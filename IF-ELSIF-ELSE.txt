SET SERVEROUTPUT ON;

DECLARE
    year1 NUMBER;
BEGIN
    year1 := &year1;

    IF MOD(year1, 400) = 0 THEN
        DBMS_OUTPUT.PUT_LINE('It is a Leap Year');
    ELSIF MOD(year1, 100) = 0 THEN
        DBMS_OUTPUT.PUT_LINE('It is not a Leap Year');
    ELSIF MOD(year1, 4) = 0 THEN
        DBMS_OUTPUT.PUT_LINE('It is a Leap Year');
    ELSE
        DBMS_OUTPUT.PUT_LINE('It is not a Leap Year');
    END IF;
END;
/

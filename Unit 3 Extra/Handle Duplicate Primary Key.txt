SET SERVEROUTPUT ON;

DECLARE
    emp_id NUMBER;
BEGIN
    emp_id := &emp_id;

    INSERT INTO EMP (EMPNO, ENAME, SAL)
    VALUES (emp_id, 'TEST', 1000);

    DBMS_OUTPUT.PUT_LINE('Employee inserted successfully.');

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('Duplicate employee ID. Employee already exists.');
END;
/

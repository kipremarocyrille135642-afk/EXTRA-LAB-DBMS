SET SERVEROUTPUT ON;

CREATE OR REPLACE FUNCTION annual_salary(
    p_monthly_salary IN NUMBER
)
RETURN NUMBER
IS
    v_annual_salary NUMBER;
BEGIN
    v_annual_salary := p_monthly_salary * 12;
    RETURN v_annual_salary;
END;
/

DECLARE
    v_monthly_salary NUMBER := &v_monthly_salary;
    v_annual_salary NUMBER;
BEGIN
    v_annual_salary := annual_salary(v_monthly_salary);

    DBMS_OUTPUT.PUT_LINE('Monthly Salary = ' || v_monthly_salary);
    DBMS_OUTPUT.PUT_LINE('Annual Salary = ' || v_annual_salary);
END;
/

SET SERVEROUTPUT ON;

CREATE OR REPLACE PROCEDURE count_employees(
    p_dept_no IN NUMBER,
    p_count OUT NUMBER
)
IS
BEGIN
    SELECT COUNT(*)
    INTO p_count
    FROM EMP
    WHERE DEPTNO = p_dept_no;
END;
/

DECLARE
    v_count NUMBER;
BEGIN
    count_employees(10, v_count);

    DBMS_OUTPUT.PUT_LINE('Number of Employees = ' || v_count);
END;
/

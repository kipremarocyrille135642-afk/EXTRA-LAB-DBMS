SET SERVEROUTPUT ON;

CREATE OR REPLACE FUNCTION calculate_bonus(
    p_salary IN NUMBER
)
RETURN NUMBER
IS
    v_bonus NUMBER;
BEGIN
    v_bonus := p_salary * 10 / 100;

    RETURN v_bonus;
END;
/

DECLARE
    v_salary NUMBER := &v_salary;
    v_bonus NUMBER;
BEGIN
    v_bonus := calculate_bonus(v_salary);

    DBMS_OUTPUT.PUT_LINE('Salary = ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('Bonus = ' || v_bonus);
END;
/

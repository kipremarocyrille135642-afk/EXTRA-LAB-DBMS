
SET SERVEROUTPUT ON;

DECLARE
    CURSOR emp_cursor IS
        SELECT ENAME, SAL
        FROM EMP;

    v_name EMP.ENAME%TYPE;
    v_sal EMP.SAL%TYPE;
BEGIN
    OPEN emp_cursor;

    LOOP
        FETCH emp_cursor INTO v_name, v_sal;
        EXIT WHEN emp_cursor%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE('Employee Name: ' || v_name ||
                             '  Salary: ' || v_sal);
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Number of employees processed: ' ||
                         emp_cursor%ROWCOUNT);

    CLOSE emp_cursor;
END;
/

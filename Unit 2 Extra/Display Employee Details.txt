SET SERVEROUTPUT ON;

DECLARE
    CURSOR emp_cursor IS
        SELECT ENAME, DEPTNO, SAL
        FROM EMP;
BEGIN
    FOR emp IN emp_cursor LOOP
        DBMS_OUTPUT.PUT_LINE(
            'Employee Name: ' || emp.ENAME ||
            '  Department Number: ' || emp.DEPTNO ||
            '  Salary: ' || emp.SAL
        );
    END LOOP;
END;

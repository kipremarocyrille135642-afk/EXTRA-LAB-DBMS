SET SERVEROUTPUT ON;

DECLARE
    ch CHAR(1);

    CURSOR emp_cursor IS
        SELECT ENAME, SAL
        FROM EMP
        WHERE ENAME LIKE ch || '%';
BEGIN
    ch := '&ch';

    FOR emp IN emp_cursor LOOP
        DBMS_OUTPUT.PUT_LINE(
            'Employee Name: ' || emp.ENAME ||
            '  Salary: ' || emp.SAL
        );
    END LOOP;
END;

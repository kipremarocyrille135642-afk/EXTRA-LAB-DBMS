SET SERVEROUTPUT ON;

DECLARE
    n NUMBER;
BEGIN
    n := &n;

    IF n > 0 THEN
        DBMS_OUTPUT.PUT_LINE('The number is Positive');
    ELSIF n < 0 THEN
        DBMS_OUTPUT.PUT_LINE('The number is Negative');
    ELSE
        DBMS_OUTPUT.PUT_LINE('The number is Zero');
    END IF;
END;
/

SET SERVEROUTPUT ON;

CREATE TABLE DELETE_EMP_AUDIT (
    EMPNO NUMBER,
    ENAME VARCHAR2(50),
    DEPTNO NUMBER,
    SAL NUMBER,
    DELETE_DATE DATE
);

CREATE OR REPLACE TRIGGER delete_emp_audit_trigger
AFTER DELETE ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO DELETE_EMP_AUDIT
    VALUES (
        :OLD.EMPNO,
        :OLD.ENAME,
        :OLD.DEPTNO,
        :OLD.SAL,
        SYSDATE
    );
END;
/

DELETE FROM EMP
WHERE EMPNO = 7369;

COMMIT;

SELECT * FROM DELETE_EMP_AUDIT;

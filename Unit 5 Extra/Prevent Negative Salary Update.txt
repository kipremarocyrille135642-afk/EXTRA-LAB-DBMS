SET SERVEROUTPUT ON;

CREATE OR REPLACE TRIGGER check_salary_update
BEFORE UPDATE OF SAL ON EMP
FOR EACH ROW
BEGIN
    IF :NEW.SAL < 0 THEN
        RAISE_APPLICATION_ERROR(-20002, 'Salary cannot be negative.');
    END IF;
END;
/

UPDATE EMP
SET SAL = -500
WHERE EMPNO = 7369;

COMMIT;

SET SERVEROUTPUT ON;

ALTER TABLE EMP
ADD (
    INSERT_DATE DATE,
    INSERT_USER VARCHAR2(50)
);

CREATE OR REPLACE TRIGGER employee_insert_info
BEFORE INSERT ON EMP
FOR EACH ROW
BEGIN
    :NEW.INSERT_DATE := SYSDATE;
    :NEW.INSERT_USER := USER;
END;
/

INSERT INTO EMP (EMPNO, ENAME, DEPTNO, SAL)
VALUES (8003, 'TEST', 10, 25000);

COMMIT;

SELECT EMPNO, ENAME, INSERT_DATE, INSERT_USER
FROM EMP
WHERE EMPNO = 8003;

SET SERVEROUTPUT ON;

CREATE OR REPLACE TRIGGER salary_change_trigger
AFTER UPDATE OF SAL ON EMP
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE('Employee ID: ' || :OLD.EMPNO);
    DBMS_OUTPUT.PUT_LINE('Old Salary: ' || :OLD.SAL);
    DBMS_OUTPUT.PUT_LINE('New Salary: ' || :NEW.SAL);
END;
/

UPDATE EMP
SET SAL = SAL + 1000
WHERE EMPNO = 7369;

COMMIT;

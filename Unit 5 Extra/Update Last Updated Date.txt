SET SERVEROUTPUT ON;

ALTER TABLE EMP
ADD LAST_UPDATED DATE;

CREATE OR REPLACE TRIGGER update_last_updated
BEFORE UPDATE ON EMP
FOR EACH ROW
BEGIN
    :NEW.LAST_UPDATED := SYSDATE;
END;
/

UPDATE EMP
SET SAL = SAL + 1000
WHERE EMPNO = 7369;

COMMIT;

SELECT EMPNO, ENAME, SAL, LAST_UPDATED
FROM EMP
WHERE EMPNO = 7369;

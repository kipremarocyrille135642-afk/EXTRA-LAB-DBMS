
SET SERVEROUTPUT ON;

CREATE TABLE EMP_AUDIT (
    EMPNO NUMBER,
    ENAME VARCHAR2(50),
    DEPTNO NUMBER,
    SAL NUMBER,
    INSERT_DATE DATE
);

CREATE OR REPLACE TRIGGER emp_audit_trigger
AFTER INSERT ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO EMP_AUDIT
    VALUES (
        :NEW.EMPNO,
        :NEW.ENAME,
        :NEW.DEPTNO,
        :NEW.SAL,
        SYSDATE
    );
END;
/

INSERT INTO EMP (EMPNO, ENAME, DEPTNO, SAL)
VALUES (8001, 'JOHN', 10, 25000);

COMMIT;

SELECT * FROM EMP_AUDIT;

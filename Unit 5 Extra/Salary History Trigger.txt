SET SERVEROUTPUT ON;

CREATE TABLE SALARY_HISTORY (
    EMPNO NUMBER,
    OLD_SALARY NUMBER,
    NEW_SALARY NUMBER,
    CHANGE_DATE DATE
);

CREATE OR REPLACE TRIGGER salary_history_trigger
AFTER UPDATE OF SAL ON EMP
FOR EACH ROW
BEGIN
    INSERT INTO SALARY_HISTORY
    VALUES (
        :OLD.EMPNO,
        :OLD.SAL,
        :NEW.SAL,
        SYSDATE
    );
END;
/

UPDATE EMP
SET SAL = SAL + 1000
WHERE EMPNO = 7369;

COMMIT;

SELECT * FROM SALARY_HISTORY;
